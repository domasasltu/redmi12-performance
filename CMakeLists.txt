cmake_minimum_required(VERSION 3.28)
project(redmi12_performance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LSPLANT_STANDALONE OFF CACHE BOOL "" FORCE)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/lsplant/lsplant/src/main/jni lsplant)
add_library(dobby_static STATIC IMPORTED)
set_target_properties(dobby_static PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/third_party/dobby-prebuilt/libdobby.a")
file(GLOB XDL_SOURCES third_party/xdl/xdl/src/main/cpp/*.c)
add_library(xdl_static STATIC ${XDL_SOURCES})
target_include_directories(xdl_static PUBLIC third_party/xdl/xdl/src/main/cpp/include PRIVATE third_party/xdl/xdl/src/main/cpp)
add_library(performance SHARED native/module.cpp native/icon.cpp native/radio.cpp)
target_include_directories(performance PRIVATE native third_party/zygisk third_party/dobby-prebuilt)
target_link_libraries(performance PRIVATE lsplant_static dobby_static xdl_static log dl)
